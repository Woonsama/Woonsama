// Woonsama.hlsl
float3 Woonsama(float2 uv)
{
    float3 art  = Artist(uv);      // 룩, 컬러, 연출을 보는 눈
    float3 tech = Programmer(uv);  // 그걸 실시간으로 구현하는 손
    return lerp(art, tech, 0.5);   // 둘 사이 어딘가에 있는 사람
}
